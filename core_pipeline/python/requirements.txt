# Pin policy: exact-pin the handful of packages whose model-loading code assumes a specific major
# API shape (torch/torchvision/onnxruntime CUDA provider behavior, transformers' tied-weights
# internals patched in ml_region_lib.py -- see the guard added in this same round), floor-pin
# (>=) the rest.

# install from the PyTorch CUDA 12.4 index, not plain PyPI, e.g.:
#   pip install torch==2.6.0 torchvision==0.21.0 --index-url https://download.pytorch.org/whl/cu124

# sentencepiece: needed directly by the NLLB local-translator tokenizer
# (facebook/nllb-200-distilled-600M). protobuf: sentencepiece needs a compatible protobuf
# to actually convert a slow tokenizer to fast at runtime -- without a matching pair,
# transformers raises "Couldn't instantiate the backend tokenizer... need sentencepiece
# or tiktoken installed" even with sentencepiece present and importable (a known
# transformers issue class, e.g. huggingface/transformers#36322). Confirmed live: this
# exact error also hits magi's bundled TrOCR/RobertaTokenizer dependency
# (microsoft/trocr-base-printed), not just NLLB -- pin both together, not just one.

protobuf>=3.20.0,<6
